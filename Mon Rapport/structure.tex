%----------------------------------------------------------------------------------------
%	CHAPTER HEADINGS (Numéro dans carré bleu)
%----------------------------------------------------------------------------------------

\makeatletter
\def\@makechapterhead#1{%
	\begin{tikzpicture}[remember picture,overlay]
	\node at (current page.north west){
		\begin{tikzpicture}[remember picture,overlay]

		% Carré contenant "Chapitre" + numéro
		\node[
		draw=myblue,
		fill=myblue,
		rounded corners=6pt,
		minimum width=3cm,
		minimum height=3cm,
		align=center,
		anchor=north west,
		text=white
		] (chapnum) at (2cm,-3cm) {
			\small\sffamily Chapitre \\[5pt]
			\fontsize{60}{70}\selectfont\bfseries \thechapter
		};

		% Titre à droite du carré
		\node[
		anchor=west,
		align=left,
		color=myblue
		] at ([xshift=1cm]chapnum.east) {
			\huge\sffamily\bfseries #1
		};

		\end{tikzpicture}};
	\end{tikzpicture}
	\vspace*{4cm}
}
\makeatother
